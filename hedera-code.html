🔗 Hedera Payment Flow

1
402 Challenge — Server issues HTTP 402 with PAYMENT-REQUIRED header containing Hedera Testnet payment requirements (receiver account, tinybars amount, facilitator fee-payer).
2
Client Signing — HederaTestnetSigner constructs a TransferTransaction anchored to the facilitator's fee-payer, adds debit/credit transfers, freezes & signs with customer's private key.
3
Facilitator Verify — FacilitatorService dispatches the signed payload to /verify to validate transaction signature and constraints.
4
Facilitator Settle — FacilitatorService dispatches to /settle — facilitator co-signs the fee and submits the CryptoTransfer to Hedera Testnet consensus.
5
Mirror Node Reconciliation — ReconciliationService independently queries the Hedera Mirror Node to confirm SUCCESS status and verify exact tinybars credited to receiver.
6
Content Unlock — Only after on-chain confirmation, the AI-generated content is released to the user with a HashScan transaction proof.

Runtime configuration with Zod validation — enforces Hedera Testnet-only operation and rejects any accidental mainnet configuration.
import dotenv from "dotenv";
import path from "path";
import { z } from "zod";
import {
  DEFAULT_FACILITATOR_URL,
  DEFAULT_MIRROR_NODE_URL,
  DEFAULT_TIMEOUT_SECONDS,
  HEDERA_TESTNET_CAIP2,
} from "./constants";

dotenv.config({ path: path.resolve(process.cwd(), ".env") });

const configSchema = z.object({
  PORT: z.coerce.number().int().positive().default(4020),
  HEDERA_NETWORK: z.string().default(HEDERA_TESTNET_CAIP2),
  HEDERA_RECEIVER_ACCOUNT_ID: z
    .string()
    .regex(/^0\.0\.[1-9][0-9]*$/, "Receiver account must follow Hedera 0.0.x format")
    .default("0.0.9999999"),
  FACILITATOR_URL: z.string().url().default(DEFAULT_FACILITATOR_URL),
  FASTAPI_GATEWAY_URL: z.string().url().default("http://localhost:8000"),
  INTERNAL_SERVICE_SECRET: z.string().min(8).default("perpay_internal_secret_dev_mode_only"),
  HEDERA_MIRROR_NODE_URL: z.string().url().default(DEFAULT_MIRROR_NODE_URL),
  SQLITE_DB_PATH: z.string().default("./data/perpay_payments.db"),
  MAX_PAYMENT_TIMEOUT_SECONDS: z.coerce.number().int().positive().default(DEFAULT_TIMEOUT_SECONDS),
  TESTNET_SMOKE_PAYER_ID: z.string().optional(),
  TESTNET_SMOKE_PAYER_KEY: z.string().optional(),
});

function loadConfig() {
  const parsed = configSchema.safeParse(process.env);
  if (!parsed.success) {
    console.error("Configuration validation error:", parsed.error.format());
    throw new Error(`Invalid configuration: ${JSON.stringify(parsed.error.format())}`);
  }

  const cfg = parsed.data;

  // Strict Mainnet rejection
  if (cfg.HEDERA_NETWORK !== HEDERA_TESTNET_CAIP2 || cfg.HEDERA_NETWORK.toLowerCase().includes("mainnet")) {
    throw new Error(
      `Accidental mainnet configuration rejected: "${cfg.HEDERA_NETWORK}". PerPay payment service strictly requires "${HEDERA_TESTNET_CAIP2}".`
    );
  }

  // Ensure mirror node is not pointed to mainnet
  if (cfg.HEDERA_MIRROR_NODE_URL.toLowerCase().includes("mainnet-public")) {
    throw new Error("Hedera mirror node cannot point to mainnet in testnet environment.");
  }

  return cfg;
}

export const config = loadConfig();
export type Config = typeof config;

Core Hedera signing class — uses the @x402/hedera SDK to construct, freeze, and sign TransferTransaction instances anchored to the facilitator's fee-payer account.
import {
  AccountId,
  Client,
  Hbar,
  PrivateKey,
  TransactionId,
  TransferTransaction,
} from "@x402/hedera";
import { HEDERA_TESTNET_CAIP2, SCHEME_EXACT, X402_VERSION } from "../constants";
import { PaymentPayloadDoc, PaymentRequirementDoc } from "../types";

export interface SignerOptions {
  payerAccountId: string;
  payerPrivateKey: string; // DER or Hex string (never committed or exposed)
}

export class HederaTestnetSigner {
  private readonly payerAccountId: AccountId;
  private readonly payerPrivateKey: PrivateKey;
  private readonly client: Client;

  constructor(options: SignerOptions) {
    if (!options.payerAccountId || !options.payerPrivateKey) {
      throw new Error("HederaTestnetSigner requires payerAccountId and payerPrivateKey");
    }

    this.payerAccountId = AccountId.fromString(options.payerAccountId);
    this.payerPrivateKey = PrivateKey.fromString(options.payerPrivateKey);
    this.client = Client.forTestnet();
  }

  /**
   * Constructs, freezes, and signs an exact cryptoTransfer transaction
   * using the facilitator's feePayer.
   */
  public async createSignedPaymentPayload(
    requirements: PaymentRequirementDoc["accepts"][0]
  ): Promise<PaymentPayloadDoc> {
    if (requirements.network !== HEDERA_TESTNET_CAIP2) {
      throw new Error(`Unsupported network for Hedera Testnet Signer: ${requirements.network}`);
    }

    if (!requirements.extra?.feePayer) {
      throw new Error("Missing mandatory extra.feePayer in payment requirements");
    }

    const feePayerId = AccountId.fromString(requirements.extra.feePayer);
    const receiverId = AccountId.fromString(requirements.payTo);
    const amountTinybars = BigInt(requirements.amount);

    if (amountTinybars <= 0n) {
      throw new Error(`Tinybars amount must be strictly positive: ${amountTinybars}`);
    }

    // Generate transaction ID anchored to the facilitator's fee payer
    const txId = TransactionId.generate(feePayerId);

    // Build cryptoTransfer: Customer account debited, receiver credited
    const transferTx = new TransferTransaction()
      .setTransactionId(txId)
      .addHbarTransfer(this.payerAccountId, Hbar.fromTinybars(-Number(amountTinybars)))
      .addHbarTransfer(receiverId, Hbar.fromTinybars(Number(amountTinybars)))
      .freezeWith(this.client);

    // Customer signs their debit transfer
    const signedTx = await transferTx.sign(this.payerPrivateKey);
    const signedBytesB64 = Buffer.from(signedTx.toBytes()).toString("base64");

    return {
      x402Version: X402_VERSION,
      scheme: SCHEME_EXACT,
      network: HEDERA_TESTNET_CAIP2,
      payload: {
        transaction: signedBytesB64,
        payer: this.payerAccountId.toString(),
      },
    };
  }

  /**
   * Helper to format the payment signature header value directly.
   */
  public async createPaymentSignatureHeader(
    requirements: PaymentRequirementDoc["accepts"][0]
  ): Promise<string> {
    const payloadDoc = await this.createSignedPaymentPayload(requirements);
    return Buffer.from(JSON.stringify(payloadDoc)).toString("base64");
  }

  public close(): void {
    this.client.close();
  }
}

Central orchestrator for the x402 pay-to-unlock flow on Hedera Testnet — builds payment requirements, issues 402 challenges, validates signatures, settles via facilitator, reconciles on Hedera Mirror Node, and unlocks content.
import crypto from "crypto";
import {
  HEDERA_NATIVE_ASSET,
  HEDERA_TESTNET_CAIP2,
  SCHEME_EXACT,
  X402_VERSION,
} from "../constants";
import { PaymentRepository } from "../db/repository";
import {
  PaymentPayloadDoc,
  PaymentRequirementDoc,
  SettlementRecord,
  SharedPaymentEvent,
  TrustedInvoiceContext,
} from "../types";
import { FacilitatorService } from "./facilitator.service";
import { GatewayClient } from "./gateway.client";
import { ReconciliationService } from "./reconciliation.service";
import { PaymentStateMachine } from "./state-machine.service";

export interface PayToUnlockResult {
  statusCode: 200 | 400 | 402 | 403 | 404 | 500;
  headers: Record<string, string>;
  body: Record<string, unknown>;
  responseUnlocked: boolean;
}

export class PaymentService {
  constructor(
    private readonly repo: PaymentRepository,
    private readonly facilitatorService: FacilitatorService,
    private readonly reconciliationService: ReconciliationService,
    private readonly gatewayClient: GatewayClient,
    private readonly receiverAccountId: string,
    private readonly maxTimeoutSeconds: number
  ) {}

  /**
   * Builds x402 v2 payment requirements document for an invoice.
   */
  public async buildPaymentRequirements(
    invoice: TrustedInvoiceContext,
    resourceUrl: string
  ): Promise<PaymentRequirementDoc> {
    const supported = await this.facilitatorService.getSupported();
    const feePayer = supported.feePayer;

    return {
      x402Version: X402_VERSION,
      accepts: [
        {
          scheme: SCHEME_EXACT,
          network: HEDERA_TESTNET_CAIP2,
          asset: HEDERA_NATIVE_ASSET,
          amount: invoice.charge_tinybars.toString(),
          payTo: this.receiverAccountId,
          maxTimeoutSeconds: this.maxTimeoutSeconds,
          extra: {
            feePayer,
            assetTransferMethod: "cryptoTransfer",
          },
        },
      ],
      resource: {
        url: resourceUrl,
        description: `PerPay AI Generation (${invoice.model}, ${invoice.total_tokens} tokens)`,
      },
    };
  }

  /**
   * Evaluates or executes the Pay-to-Unlock flow for a generation result.
   */
  public async handlePayToUnlock(
    generationId: string,
    resourceUrl: string,
    paymentSignatureHeader?: string,
    authenticatedUserId?: string
  ): Promise<PayToUnlockResult> {
    // 1. Check if generation has already been settled and confirmed in durable local DB
    const existingSettlement = this.repo.getSettlementByGenerationId(generationId);
    if (existingSettlement) {
      const storedContent = await this.gatewayClient.fetchStoredResponse(generationId);
      return {
        statusCode: 200,
        headers: { "Cache-Control": "private, no-cache" },
        body: {
          generation_id: generationId,
          content: storedContent?.content || "(Response already unlocked)",
          model: storedContent?.model,
          total_tokens: storedContent?.total_tokens,
          settlement: {
            transaction_id: existingSettlement.transaction_id,
            transaction_url: existingSettlement.hashscan_url,
            amount_tinybars: existingSettlement.amount_tinybars,
            confirmed_at: existingSettlement.confirmed_at,
          },
        },
        responseUnlocked: true,
      };
    }

    // 2. Fetch authoritative invoice context from FastAPI
    let invoice = this.repo.getInvoiceByGenerationId(generationId);
    if (!invoice) {
      invoice = await this.gatewayClient.fetchInvoiceContext(generationId);
      if (invoice) { this.repo.saveInvoice(invoice); }
    }

    if (!invoice) {
      return {
        statusCode: 404, headers: {},
        body: { error: `No active invoice found for generation ${generationId}` },
        responseUnlocked: false,
      };
    }

    // Check account-level ownership
    if (authenticatedUserId && invoice.user_id !== authenticatedUserId) {
      return {
        statusCode: 403, headers: {},
        body: { error: "User unauthorized to access this generation invoice" },
        responseUnlocked: false,
      };
    }

    // Check invoice expiry
    if (new Date(invoice.expires_at).getTime() < Date.now()) {
      return {
        statusCode: 400, headers: {},
        body: {
          error: "Invoice has expired. Please create a new generation request.",
          invoice_id: invoice.invoice_id,
          expires_at: invoice.expires_at,
        },
        responseUnlocked: false,
      };
    }

    // 3. If no PAYMENT-SIGNATURE header provided, issue HTTP 402 challenge
    const requirementDoc = await this.buildPaymentRequirements(invoice, resourceUrl);
    const requirementB64 = Buffer.from(JSON.stringify(requirementDoc)).toString("base64");

    if (!paymentSignatureHeader) {
      return {
        statusCode: 402,
        headers: {
          "PAYMENT-REQUIRED": requirementB64,
          "Content-Type": "application/json",
        },
        body: {
          error: "Payment Required",
          invoice_id: invoice.invoice_id,
          generation_id: invoice.generation_id,
          charge_tinybars: invoice.charge_tinybars,
          charge_hbar: invoice.charge_hbar,
          network: HEDERA_TESTNET_CAIP2,
          payTo: this.receiverAccountId,
          feePayer: requirementDoc.accepts[0].extra.feePayer,
        },
        responseUnlocked: false,
      };
    }

    // 4. Decode and validate PAYMENT-SIGNATURE payload
    let payloadDoc: PaymentPayloadDoc;
    try {
      const decodedJson = Buffer.from(paymentSignatureHeader, "base64").toString("utf-8");
      payloadDoc = JSON.parse(decodedJson);
    } catch {
      return {
        statusCode: 400,
        headers: { "PAYMENT-REQUIRED": requirementB64 },
        body: { error: "Invalid PAYMENT-SIGNATURE header: failed to decode base64 JSON" },
        responseUnlocked: false,
      };
    }

    if (
      payloadDoc.x402Version !== X402_VERSION ||
      payloadDoc.scheme !== SCHEME_EXACT ||
      payloadDoc.network !== HEDERA_TESTNET_CAIP2 ||
      !payloadDoc.payload?.transaction
    ) {
      return {
        statusCode: 400,
        headers: { "PAYMENT-REQUIRED": requirementB64 },
        body: {
          error: "Unsupported x402 payment signature format or network",
          expected: { x402Version: 2, scheme: "exact", network: HEDERA_TESTNET_CAIP2 },
        },
        responseUnlocked: false,
      };
    }

    // 5. Replay and Idempotency Check
    const digest = crypto.createHash("sha256").update(payloadDoc.payload.transaction).digest("hex");
    const paymentId = crypto.randomUUID();
    this.repo.recordPaymentAttempt({
      payment_id: paymentId, invoice_id: invoice.invoice_id,
      generation_id: invoice.generation_id, payload_digest: digest,
      raw_payload: JSON.stringify(payloadDoc), state: "PAYMENT_SUBMITTED",
      created_at: new Date().toISOString(), updated_at: new Date().toISOString(),
    });

    PaymentStateMachine.assertValidTransition("PAYMENT_REQUIRED", "PAYMENT_SUBMITTED");

    // 6. Verification with Facilitator
    PaymentStateMachine.assertValidTransition("PAYMENT_SUBMITTED", "VERIFYING");
    this.repo.updatePaymentAttemptState(paymentId, "VERIFYING");

    const reqAccept = requirementDoc.accepts[0];
    const verifyResult = await this.facilitatorService.verifyPayment(payloadDoc, reqAccept);
    if (!verifyResult.isValid) {
      this.repo.updatePaymentAttemptState(paymentId, "FAILED", verifyResult.invalidReason);
      return {
        statusCode: 402,
        headers: { "PAYMENT-REQUIRED": requirementB64 },
        body: { error: "Payment verification failed", reason: verifyResult.invalidReason },
        responseUnlocked: false,
      };
    }

    // 7. Settlement with Facilitator
    PaymentStateMachine.assertValidTransition("VERIFYING", "SETTLING");
    this.repo.updatePaymentAttemptState(paymentId, "SETTLING");

    const settleResult = await this.facilitatorService.settlePayment(payloadDoc, reqAccept);
    if (!settleResult.success || !settleResult.transactionId) {
      this.repo.updatePaymentAttemptState(paymentId, "FAILED", settleResult.error);
      return {
        statusCode: 402,
        headers: { "PAYMENT-REQUIRED": requirementB64 },
        body: { error: "Payment settlement failed", reason: settleResult.error },
        responseUnlocked: false,
      };
    }

    const txId = settleResult.transactionId;

    // Duplicate transaction check
    const existingTx = this.repo.getSettlementByTransactionId(txId);
    if (existingTx && existingTx.invoice_id !== invoice.invoice_id) {
      this.repo.updatePaymentAttemptState(paymentId, "FAILED",
        `Transaction ${txId} already applied to another invoice ${existingTx.invoice_id}`);
      return {
        statusCode: 400,
        headers: { "PAYMENT-REQUIRED": requirementB64 },
        body: { error: "Duplicate transaction rejected. Transaction already redeemed." },
        responseUnlocked: false,
      };
    }

    // 8. Independent On-Chain Reconciliation via Hedera Mirror Node
    PaymentStateMachine.assertValidTransition("SETTLING", "RECONCILIATION_REQUIRED");
    this.repo.updatePaymentAttemptState(paymentId, "RECONCILIATION_REQUIRED");

    const recon = await this.reconciliationService.reconcileOnChain(
      txId, this.receiverAccountId, invoice.charge_tinybars, payloadDoc.payload.payer
    );

    if (!recon.isConfirmed) {
      this.repo.updatePaymentAttemptState(paymentId, "SETTLEMENT_PENDING", recon.errorMessage);
      return {
        statusCode: 402,
        headers: { "PAYMENT-REQUIRED": requirementB64 },
        body: {
          error: "Settlement submitted but pending on-chain confirmation. Please retry.",
          transaction_id: txId,
          transaction_url: recon.hashscanUrl,
        },
        responseUnlocked: false,
      };
    }

    // 9. Confirmed Settlement: Atomic DB record & Outbox Event
    PaymentStateMachine.assertValidTransition("RECONCILIATION_REQUIRED", "CONFIRMED");
    const confirmedSettlement: SettlementRecord = {
      transaction_id: recon.transactionId,
      invoice_id: invoice.invoice_id,
      generation_id: invoice.generation_id,
      payer_account: recon.payerAccount,
      receiver_account: recon.receiverAccount,
      amount_tinybars: recon.amountTinybars,
      consensus_timestamp: recon.consensusTimestamp,
      mirror_status: recon.mirrorStatus,
      hashscan_url: recon.hashscanUrl,
      confirmed_at: new Date().toISOString(),
    };

    const sharedEvent: SharedPaymentEvent = {
      schema_version: "2.0.0",
      event_id: crypto.randomUUID(),
      timestamp: confirmedSettlement.confirmed_at,
      request_id: invoice.request_id,
      invoice_id: invoice.invoice_id,
      generation_id: invoice.generation_id,
      user_id: invoice.user_id,
      wallet: recon.payerAccount || null,
      model: invoice.model,
      prompt_tokens: invoice.prompt_tokens,
      completion_tokens: invoice.completion_tokens,
      total_tokens: invoice.total_tokens,
      charge_tinybars: invoice.charge_tinybars,
      charge_hbar: invoice.charge_hbar,
      network: HEDERA_TESTNET_CAIP2,
      payment_scheme: SCHEME_EXACT,
      payment_status: "CONFIRMED",
      transaction_id: recon.transactionId,
      transaction_url: recon.hashscanUrl,
      delivery_status: "DELIVERED",
    };

    this.repo.saveConfirmedSettlement(confirmedSettlement, sharedEvent);
    this.repo.updatePaymentAttemptState(paymentId, "CONFIRMED");

    this.gatewayClient.notifySettlement(confirmedSettlement).catch(() => {});

    // 10. Fetch stored response and unlock
    const storedResponse = await this.gatewayClient.fetchStoredResponse(invoice.generation_id);

    const paymentResponsePayload = {
      success: true,
      transaction: recon.transactionId,
      transactionUrl: recon.hashscanUrl,
      network: HEDERA_TESTNET_CAIP2,
      amount: invoice.charge_tinybars.toString(),
      payer: recon.payerAccount,
    };
    const paymentResponseHeader = Buffer.from(JSON.stringify(paymentResponsePayload)).toString("base64");

    return {
      statusCode: 200,
      headers: {
        "PAYMENT-RESPONSE": paymentResponseHeader,
        "Cache-Control": "private, no-cache",
      },
      body: {
        generation_id: invoice.generation_id,
        content: storedResponse?.content || "(Response unlocked)",
        model: storedResponse?.model || invoice.model,
        total_tokens: storedResponse?.total_tokens || invoice.total_tokens,
        settlement: {
          transaction_id: recon.transactionId,
          transaction_url: recon.hashscanUrl,
          amount_tinybars: recon.amountTinybars,
          confirmed_at: confirmedSettlement.confirmed_at,
        },
      },
      responseUnlocked: true,
    };
  }

  DEFAULT_FACILITATOR_FEE_PAYER,
  HEDERA_TESTNET_CAIP2,
  SCHEME_EXACT,
  X402_VERSION,
} from "../constants";
import {
  FacilitatorSupportedKind,
  FacilitatorSupportedResponse,
  PaymentPayloadDoc,
  PaymentRequirementDoc,
} from "../types";

export interface FacilitatorCheckResult {
  isCompatible: boolean;
  feePayer: string;
  kind?: FacilitatorSupportedKind;
  errorMessage?: string;
}

export class FacilitatorService {
  private cachedFeePayer: string | null = null;
  private lastCheckedTimestamp = 0;
  private readonly CACHE_TTL_MS = 60_000; // 1 minute

  constructor(private readonly facilitatorUrl: string) {}

  public getFacilitatorUrl(): string {
    return this.facilitatorUrl;
  }

  /**
   * Queries /supported and parses facilitator compatibility for Hedera Testnet x402 exact.
   */
  public async getSupported(): Promise<FacilitatorCheckResult> {
    const now = Date.now();
    if (this.cachedFeePayer && now - this.lastCheckedTimestamp < this.CACHE_TTL_MS) {
      return { isCompatible: true, feePayer: this.cachedFeePayer };
    }

    try {
      const resp = await fetch(`${this.facilitatorUrl.replace(/\/+$/, "")}/supported`, {
        method: "GET",
        headers: { Accept: "application/json" },
        signal: AbortSignal.timeout(5000),
      });

      if (!resp.ok) {
        return {
          isCompatible: false,
          feePayer: DEFAULT_FACILITATOR_FEE_PAYER,
          errorMessage: `Facilitator returned HTTP ${resp.status} on /supported`,
        };
      }

      const body = (await resp.json()) as FacilitatorSupportedResponse;
      const parsed = this.parseSupportedResponse(body);

      if (parsed.isCompatible && parsed.feePayer) {
        this.cachedFeePayer = parsed.feePayer;
        this.lastCheckedTimestamp = now;
      }

      return parsed;
    } catch (err: unknown) {
      const msg = err instanceof Error ? err.message : String(err);
      return {
        isCompatible: false,
        feePayer: DEFAULT_FACILITATOR_FEE_PAYER,
        errorMessage: `Failed to query facilitator /supported: ${msg}`,
      };
    }
  }

  /**
   * Pure parser validating facilitator supported kind for hedera:testnet exact.
   */
  public parseSupportedResponse(response: FacilitatorSupportedResponse): FacilitatorCheckResult {
    if (!response || !Array.isArray(response.kinds)) {
      return {
        isCompatible: false,
        feePayer: DEFAULT_FACILITATOR_FEE_PAYER,
        errorMessage: "Invalid facilitator response format: missing kinds array",
      };
    }

    const matchingKind = response.kinds.find(
      (k) =>
        k.x402Version === X402_VERSION &&
        k.scheme === SCHEME_EXACT &&
        k.network === HEDERA_TESTNET_CAIP2
    );

    if (!matchingKind) {
      return {
        isCompatible: false,
        feePayer: DEFAULT_FACILITATOR_FEE_PAYER,
        errorMessage: `Facilitator does not support x402Version 2 exact scheme on ${HEDERA_TESTNET_CAIP2}`,
      };
    }

    const feePayer =
      matchingKind.extra?.feePayer ||
      (response.signers && response.signers["hedera:*"]?.[0]) ||
      DEFAULT_FACILITATOR_FEE_PAYER;

    if (!/^0\.0\.[1-9][0-9]*$/.test(feePayer)) {
      return {
        isCompatible: false,
        feePayer: DEFAULT_FACILITATOR_FEE_PAYER,
        errorMessage: `Facilitator advertised invalid Hedera fee-payer account ID: "${feePayer}"`,
      };
    }

    return { isCompatible: true, feePayer, kind: matchingKind };
  }

  /**
   * Dispatches payment payload to facilitator /verify endpoint.
   */
  public async verifyPayment(
    paymentPayload: PaymentPayloadDoc,
    requirements: PaymentRequirementDoc["accepts"][0]

      const resp = await fetch(`${this.facilitatorUrl.replace(/\/+$/, "")}/verify`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          x402Version: X402_VERSION,
          paymentPayload,
          paymentRequirements: requirements,
        }),
        signal: AbortSignal.timeout(8000),
      });

      if (!resp.ok) {
        const text = await resp.text();
        return { isValid: false, invalidReason: `Facilitator verify returned HTTP ${resp.status}: ${text}` };
      }

      const data = (await resp.json()) as { isValid: boolean; invalidReason?: string };
      return { isValid: !!data.isValid, invalidReason: data.invalidReason };
    } catch (err: unknown) {
      const msg = err instanceof Error ? err.message : String(err);
      return { isValid: false, invalidReason: `Facilitator verification call failed: ${msg}` };
    }
  }

  /**
   * Dispatches payment payload to facilitator /settle endpoint to submit on Hedera Testnet.
   */
  public async settlePayment(
    paymentPayload: PaymentPayloadDoc,
    requirements: PaymentRequirementDoc["accepts"][0]
  ): Promise<{ success: boolean; transactionId?: string; error?: string }> {
    try {
      const resp = await fetch(`${this.facilitatorUrl.replace(/\/+$/, "")}/settle`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          x402Version: X402_VERSION,
          paymentPayload,
          paymentRequirements: requirements,
        }),
        signal: AbortSignal.timeout(15000),
      });

      if (!resp.ok) {
        const text = await resp.text();
        return { success: false, error: `Facilitator settlement returned HTTP ${resp.status}: ${text}` };
      }

      const data = (await resp.json()) as {
        success: boolean; transaction?: string; transactionId?: string; error?: string;
      };

      const txId = data.transactionId || data.transaction;
      return { success: !!data.success && !!txId, transactionId: txId, error: data.error };
    } catch (err: unknown) {
      const msg = err instanceof Error ? err.message : String(err);
      return { success: false, error: `Facilitator settlement network error: ${msg}` };
    }
  }

  HASHSCAN_TESTNET_BASE_URL,
  MAX_RECONCILIATION_ATTEMPTS,
  RECONCILIATION_DELAY_MS,
} from "../constants";

export interface HederaTransferItem {
  account: string;
  amount: number;
  is_approval?: boolean;
}

export interface HederaTransactionResponse {

    result: string;
    consensus_timestamp: string;
    transaction_id: string;
    name: string;
    transfers: HederaTransferItem[];
    charged_tx_fee?: number;
  }>;

export interface ReconciliationResult {
  isConfirmed: boolean;
  transactionId: string;
  consensusTimestamp: string;
  payerAccount: string;
  receiverAccount: string;
  amountTinybars: number;
  mirrorStatus: string;
  hashscanUrl: string;
  errorMessage?: string;
}

export class ReconciliationService {
  constructor(private readonly mirrorNodeBaseUrl: string) {}

  /**
   * Normalizes Hedera transaction ID to canonical Mirror Node / HashScan format.
   * e.g., "0.0.123@1700000000.123456789" -> "0.0.123-1700000000-123456789"
   */
  public normalizeTransactionId(rawId: string): string {
    const trimmed = rawId.trim();
    if (trimmed.includes("@")) {
      const [account, timePart] = trimmed.split("@");
      const [seconds, nanos] = (timePart || "").split(".");
      return `${account}-${seconds}-${nanos}`;
    }
    return trimmed;
  }

  /**
   * Formats a canonical HashScan URL for Hedera Testnet.
   * Never fabricates 0x EVM hashes.
   */
  public getHashScanUrl(transactionId: string): string {
    const normalized = this.normalizeTransactionId(transactionId);
    return `${HASHSCAN_TESTNET_BASE_URL}/${normalized}`;
  }

  /**
   * Queries Hedera Mirror Node with bounded retries to handle indexing delay.
   */
  public async reconcileOnChain(
    rawTransactionId: string,
    expectedReceiver: string,
    expectedTinybars: number,
    expectedPayer?: string
  ): Promise<ReconciliationResult> {
    const normalizedTxId = this.normalizeTransactionId(rawTransactionId);
    const hashscanUrl = this.getHashScanUrl(normalizedTxId);

    let lastError = "";

    for (let attempt = 1; attempt <= MAX_RECONCILIATION_ATTEMPTS; attempt++) {
      try {
        const url = `${this.mirrorNodeBaseUrl.replace(/\/+$/, "")}/api/v1/transactions/${normalizedTxId}`;
        const resp = await fetch(url, {
          method: "GET",
          headers: { Accept: "application/json" },
          signal: AbortSignal.timeout(5000),
        });

        if (resp.status === 404) {
          lastError = `Transaction ${normalizedTxId} not yet indexed on Mirror Node (attempt ${attempt}/${MAX_RECONCILIATION_ATTEMPTS})`;
          if (attempt < MAX_RECONCILIATION_ATTEMPTS) {
            await new Promise((r) => setTimeout(r, RECONCILIATION_DELAY_MS));
            continue;
          }
          break;
        }

        if (!resp.ok) { lastError = `Mirror node returned HTTP ${resp.status}`; break; }

        const data = (await resp.json()) as HederaTransactionResponse;
        const tx = data.transactions && data.transactions[0];

        if (!tx) { lastError = "Mirror Node returned empty transactions list"; break; }

        if (tx.result !== "SUCCESS") {
          return {
            isConfirmed: false, transactionId: normalizedTxId,
            consensusTimestamp: tx.consensus_timestamp || "",
            payerAccount: expectedPayer || "", receiverAccount: expectedReceiver,
            amountTinybars: expectedTinybars, mirrorStatus: tx.result, hashscanUrl,
            errorMessage: `On-chain transaction execution failed with status: ${tx.result}`,
          };
        }

        // Verify transfers list
        const transfers = tx.transfers || [];
        const receiverCredit = transfers.find(
          (t) => t.account === expectedReceiver && t.amount === expectedTinybars
        );

        if (!receiverCredit) {
          return {
            isConfirmed: false, transactionId: normalizedTxId,
            consensusTimestamp: tx.consensus_timestamp,
            payerAccount: expectedPayer || "", receiverAccount: expectedReceiver,
            amountTinybars: expectedTinybars, mirrorStatus: tx.result, hashscanUrl,
            errorMessage: `Hedera transfers list missing expected credit of ${expectedTinybars} tinybars to receiver ${expectedReceiver}`,
          };
        }

        const payerDebit = transfers.find((t) =>
          expectedPayer
            ? t.account === expectedPayer && t.amount === -expectedTinybars
            : t.amount === -expectedTinybars
        );

        if (!payerDebit) {
          return {
            isConfirmed: false, transactionId: normalizedTxId,
            consensusTimestamp: tx.consensus_timestamp,
            payerAccount: expectedPayer || "", receiverAccount: expectedReceiver,
            amountTinybars: expectedTinybars, mirrorStatus: tx.result, hashscanUrl,
            errorMessage: `Hedera transfers list missing debit of ${expectedTinybars} tinybars from payer`,
          };
        }

        return {
          isConfirmed: true, transactionId: normalizedTxId,
          consensusTimestamp: tx.consensus_timestamp,
          payerAccount: payerDebit.account, receiverAccount: receiverCredit.account,
          amountTinybars: receiverCredit.amount, mirrorStatus: tx.result, hashscanUrl,
        };
      } catch (err: unknown) {
        lastError = err instanceof Error ? err.message : String(err);
        if (attempt < MAX_RECONCILIATION_ATTEMPTS) {
          await new Promise((r) => setTimeout(r, RECONCILIATION_DELAY_MS));
        }
      }
    }

    return {
      isConfirmed: false, transactionId: normalizedTxId,
      consensusTimestamp: "", payerAccount: expectedPayer || "",
      receiverAccount: expectedReceiver, amountTinybars: expectedTinybars,
      mirrorStatus: "INDEXING_TIMEOUT", hashscanUrl,
      errorMessage: `Reconciliation delayed: ${lastError}`,
    };
  }
}

import { TINYBARS_PER_HBAR } from "../constants";

/**
 * Validates and converts an integer tinybar count to a safe positive number.
 * Fails closed on negative, zero, non-integer, or overflowing values.
 */
export function validateTinybarAmount(tinybars: unknown): number {
  if (typeof tinybars === "bigint") {
    if (tinybars <= 0n) {
      throw new TypeError("Tinybar amount must be strictly positive.");
    }
    if (tinybars > BigInt(Number.MAX_SAFE_INTEGER)) {
      throw new RangeError("Tinybar amount exceeds JavaScript MAX_SAFE_INTEGER.");
    }
    return Number(tinybars);
  }

  if (typeof tinybars === "number") {
    if (!Number.isSafeInteger(tinybars) || tinybars <= 0) {
      throw new TypeError("Tinybar amount must be a positive safe integer.");
    }
    return tinybars;
  }

  if (typeof tinybars === "string") {
    const trimmed = tinybars.trim();
    if (!/^[1-9][0-9]*$/.test(trimmed)) {
      throw new TypeError(`Invalid integer tinybar string format: "${tinybars}"`);
    }
    const parsed = Number(trimmed);
    if (!Number.isSafeInteger(parsed)) {
      throw new RangeError("Tinybar string exceeds JavaScript safe integer range.");
    }
    return parsed;
  }

  throw new TypeError(`Unsupported tinybar amount type: ${typeof tinybars}`);
}

/**
 * Converts a string or numeric HBAR amount to exact integer tinybars.
 * 1 HBAR = 100,000,000 tinybars (8 decimal places).
 * Uses string splitting to guarantee zero floating point inaccuracies.
 */
export function hbarToTinybars(hbar: string | number): number {
  const hbarStr = typeof hbar === "number" ? hbar.toString() : hbar.trim();

  if (!/^[0-9]+(\.[0-9]+)?$/.test(hbarStr)) {
    throw new TypeError(`Invalid HBAR amount format: "${hbar}"`);
  }

  const [wholePart, fracPart = ""] = hbarStr.split(".");
  if (fracPart.length > 8) {
    throw new TypeError(`HBAR amount cannot have more than 8 decimal places: "${hbarStr}"`);
  }

  const paddedFrac = fracPart.padEnd(8, "0");
  const tinybarsBigInt = BigInt(wholePart) * TINYBARS_PER_HBAR + BigInt(paddedFrac);

  return validateTinybarAmount(tinybarsBigInt);
}

/**
 * Converts integer tinybars to exact string HBAR with up to 8 decimal places.
 */
export function tinybarsToHbar(tinybars: number | bigint): string {
  const safeTinybars = validateTinybarAmount(tinybars);
  const tbBigInt = BigInt(safeTinybars);

  const whole = tbBigInt / TINYBARS_PER_HBAR;
  const remainder = tbBigInt % TINYBARS_PER_HBAR;

  if (remainder === 0n) {
    return whole.toString();
  }

  const remainderStr = remainder.toString().padStart(8, "0");
  const trimmedRemainder = remainderStr.replace(/0+$/, "");
  return `${whole}.${trimmedRemainder}`;
}

Payment state machine governing the Hedera settlement lifecycle — enforces legal transitions from PAYMENT_REQUIRED through CONFIRMED or FAILED.
import { PaymentState } from "../types";

export class PaymentStateMachine {
  private static readonly ALLOWED_TRANSITIONS: Record<PaymentState, PaymentState[]> = {
    PAYMENT_REQUIRED: ["PAYMENT_SUBMITTED", "EXPIRED"],
    PAYMENT_SUBMITTED: ["VERIFYING", "FAILED"],
    VERIFYING: ["SETTLING", "FAILED"],
    SETTLING: ["SETTLEMENT_PENDING", "RECONCILIATION_REQUIRED", "CONFIRMED", "FAILED"],
    SETTLEMENT_PENDING: ["RECONCILIATION_REQUIRED", "CONFIRMED", "FAILED"],
    RECONCILIATION_REQUIRED: ["CONFIRMED", "FAILED"],
    CONFIRMED: [], // Terminal success state
    FAILED: ["PAYMENT_SUBMITTED"], // Can retry payment submission
    EXPIRED: [], // Terminal state
  };

  public static canTransition(from: PaymentState, to: PaymentState): boolean {
    if (from === to) { return true; }
    const allowed = this.ALLOWED_TRANSITIONS[from] || [];
    return allowed.includes(to);
  }

  public static assertValidTransition(from: PaymentState, to: PaymentState): void {
    if (!this.canTransition(from, to)) {
      throw new Error(
        `Illegal payment state transition from "${from}" to "${to}". Permitted transitions: [${(
          this.ALLOWED_TRANSITIONS[from] || []
        ).join(", ")}]`
      );
    }
  }

  public static isTerminal(state: PaymentState): boolean {
    return state === "CONFIRMED" || state === "EXPIRED";
  }

  public static permitsResponseUnlock(state: PaymentState): boolean {
    return state === "CONFIRMED";
  }
}
